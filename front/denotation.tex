\begin{denotation}[4cm]

% ---------- Abbreviations ----------

\item[PD]{
  Parkinson's Disease（帕金森氏症）
}

\item[MDS-UPDRS]{
  Movement Disorder Society -- Unified Parkinson's Disease Rating Scale
}

\item[NTUH]{
  National Taiwan University Hospital（國立臺灣大學醫學院附設醫院）
}

\item[LOOCV]{
  Leave-One-Out Cross-Validation（留一法交叉驗證）
}

\item[AUROC]{
  Area Under the Receiver Operating Characteristic curve（ROC 曲線下面積）
}

\item[TPR]{
  True Positive Rate; Sensitivity（真陽性率；敏感度）
}

\item[FPR]{
  False Positive Rate; $1 -$ Specificity（偽陽性率）
}

\item[LR]{
  Logistic Regression（邏輯斯迴歸）
}

\item[RF]{
  Random Forest（隨機森林）
}

\item[XGBoost]{
  Extreme Gradient Boosting
}

\item[ANOVA]{
  Analysis of Variance（變異數分析）
}

\item[GCN]{
  Graph Convolutional Network（圖卷積網路）
}

\item[AGCN]{
  Adaptive Graph Convolutional Network（自適應圖卷積網路）
}

\item[SGC-LL]{
  Spectral Graph Convolution with Laplacian Learning
}

\item[RMS]{
  Root Mean Square（均方根）
}

% ---------- Mathematical Symbols ----------

\item[$r(\tau)$]{
  Autocorrelation function of the detrended joint-axis trajectory at lag $\tau$
}

\item[$\tau_{\mathrm{peak}}$]{
  Lag, measured in frames, of the first positive-lag peak of $r(\tau)$
}

\item[$f_{\mathrm{rot}}$]{
  Rotation-frequency descriptor computed as $f_{\mathrm{rot}} = f_s / \tau_{\mathrm{peak}}$
}

\item[$C_{ij}^{(n)}$]{
  Pearson correlation coefficient between the feature vectors of joints $i$ and $j$ for patient $n$, $C_{ij}^{(n)} \in [-1, 1]$
}

\item[$A_{ij}^{(n)}$]{
  Remapped joint-correlation adjacency entry, $A_{ij}^{(n)} = (C_{ij}^{(n)} + 1)/2 \in [0,1]$
}

\item[$\hat{\boldsymbol{\Sigma}}^{(n)}$]{
  Ledoit--Wolf shrinkage covariance estimate for patient $n$
}

\item[$\rho_{ij}$]{
  Spearman rank correlation between adjacency entry $A_{ij}$ and MDS-UPDRS stage across all patients
}

\item[$\mathbb{D}(x_i, x_j)$]{
  Generalised Mahalanobis distance between node feature vectors $x_i$ and $x_j$ in AGCN
}

\item[$\tilde{A}_{ij}$]{
  Gaussian-kernel soft adjacency weight derived from $\mathbb{D}(x_i, x_j)$
}

\item[$L$]{
  Anatomical skeleton graph Laplacian (fixed)
}

\item[$L_{\mathrm{res}}$]{
  Residual (data-driven) graph Laplacian learned by AGCN
}

\item[$\hat{L}$]{
  Augmented graph Laplacian, $\hat{L} = L + \alpha\, L_{\mathrm{res}}$
}

\item[$\alpha$]{
  Scalar weight controlling the influence of the residual graph Laplacian in AGCN
}

\item[$W_d$]{
  Learnable projection matrix defining the Mahalanobis metric in AGCN, $W_d \in \mathbb{R}^{d \times d}$
}

\item[$J(\theta)$]{
  Youden index at classification threshold $\theta$, $J(\theta) = \mathrm{TPR}(\theta) - \mathrm{FPR}(\theta)$
}

\item[$\theta^{*}$]{
  Youden-selected validation threshold
}

\end{denotation}
